Игрок

scripts/player.lualua
local P = screw.system("Player")
P.state = { jumps_left = 2 }

function P:fixed_update(dt)
    local h = screw.input.axis("horizontal")
    for id, body, tag in P:each("RigidBody", "Tag:player") do
        body.velocity.x = h * 6.5
        if screw.input.pressed("jump") and P.state.jumps_left > 0 then
            body.velocity.y = 8.0
            P.state.jumps_left = P.state.jumps_left - 1
        end
        if body.on_ground then P.state.jumps_left = 2 end
    end
end

Камера

scripts/camera.lualua
local C = screw.system("CameraFollow")

function C:update(dt)
    local player = screw.find("Tag:player")
    local cam    = screw.find("Tag:main_camera")
    local target = player:get("Transform").position
    local cur    = cam:get("Transform").position
    cam:get("Transform").position = cur + (target - cur) * math.min(1, dt * 6)
end

Чекпоинт

scripts/checkpoint.lualua
screw.on("trigger_enter", function(zone, other)
    if screw.has_tag(zone, "checkpoint") and screw.has_tag(other, "player") then
        screw.state.spawn_point = zone:get("Transform").position
        screw.log.info("чекпоинт сохранён")
    end
end)
